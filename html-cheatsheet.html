Images & Media

images

img, figure, picture

<!-- Basic image -->
<img src="photo.jpg"
     alt="Description"
     width="800"
     height="600"
     loading="lazy">

<!-- With caption -->
<figure>
  <img src="photo.jpg" alt="...">
  <figcaption>Caption</figcaption>
</figure>

<!-- Responsive -->
<picture>
  <source media="(max-width:600px)"
          srcset="small.jpg">
  <img src="large.jpg" alt="...">
</picture>
audio / video

Multimedia Elements

<!-- Video -->
<video controls
       width="640"
       poster="thumb.jpg"
       muted autoplay loop>
  <source src="movie.mp4"
          type="video/mp4">
  <source src="movie.webm"
          type="video/webm">
  Fallback text for old browsers.
</video>

<!-- Audio -->
<audio controls>
  <source src="sound.mp3"
          type="audio/mpeg">
</audio>

<!-- Embed / iframe -->
<iframe
  src="https://example.com"
  width="600" height="400"
  title="Accessible title"
  loading="lazy">
</iframe>

Meta, SEO & OG Tags

essential meta

SEO + Social Meta Tags

<!-- Core -->
<meta charset="UTF-8">
<meta name="description"
      content="Page description">
<meta name="robots"
      content="index, follow">

<!-- Open Graph (Facebook/LinkedIn) -->
<meta property="og:title"
      content="Title">
<meta property="og:description"
      content="Desc">
<meta property="og:image"
      content="img.png">
<meta property="og:url"
      content="https://...">

<!-- Twitter Card -->
<meta name="twitter:card"
      content="summary_large_image">
<meta name="twitter:creator"
      content="@handle">

<!-- Favicon -->
<link rel="icon"
      href="/favicon.ico">
<link rel="apple-touch-icon"
      href="/apple-icon.png">

<!-- Canonical URL -->
<link rel="canonical"
      href="https://yoursite.com/page">
performance

Resource Hints

<!-- Preload critical resources -->
<link rel="preload"
      href="font.woff2"
      as="font"
      crossorigin>

<!-- Prefetch next page -->
<link rel="prefetch"
      href="/next-page">

<!-- DNS prefetch -->
<link rel="dns-prefetch"
      href="//fonts.googleapis.com">

<!-- Script loading strategies -->
<script src="app.js" defer></script>
<!-- defer: runs after HTML parsed -->

<script src="widget.js" async></script>
<!-- async: runs as soon as loaded -->

<!-- Module script -->
<script type="module" src="main.js">
</script>

data attributes

Custom data-* Attributes

<!-- Store custom data -->
<button
  data-user-id="42"
  data-action="delete">
  Delete
</button>

// Access in JS:
const btn = document.querySelector('button');
btn.dataset.userId;  // "42"
btn.dataset.action;  // "delete"

// CSS access:
content: attr(data-action);
